### Example

\*\*Paginate nested `registrations` to indices `0..9` per parent event\*\*

```ts
import type { TransformRangeOptions } from 'firebase/sql';

const rangeOpts: TransformRangeOptions = { referencedTable: 'registrations' };

  .range(0, 9, rangeOpts);
```
